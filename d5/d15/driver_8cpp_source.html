101void Driver::setup(Variables* theVars, const bool continuingRun) {
102 if (!theVars) {
103 if (currentVars) {
104 delete currentVars;
105 currentVars = nullptr;
106 }
107 currentVars = new Variables(inputVars);
108 theVars = currentVars;
109 }
110 bool dumpStep = _dumpStep;
111 if (continuingRun)
112 _dumpStep = false;
113 theVars->finalizeInputs(true);
114 // Initialize the structure of the model, i.e. Is this model separate
115 // or combined with others.
116 IniModelCom(theVars);
117 setup_connections(theVars);
118 // The time information is set in a global variable called tglobal in SYSInitial.
119 SYSInitial(theVars);
120 setup_variables(theVars);
121 arr prev_constraints = constraints;
122 if (!continuingRun) {
123 setup_param(theVars);
124 }
125 setup_constraints(theVars);
126 if (continuingRun) {
127 constraints = prev_constraints;
128 }
129 theVars->inputsFinalized = true;
130 theVars->inputsUpdated.clear();
131 if (continuingRun)
132 _dumpStep = dumpStep;
133}

135void Driver::teardown(const realtype& t, const N_Vector& y,
136 Variables* theVars) {
137 if (!theVars) theVars = currentVariables();
138 intermediateRes = N_VGetArrayPointer(y);
139 for (size_t i = 0; i < constraints.size(); i++)
140 constraints[i] = intermediateRes[i];
141 _lastStep = true;
142 _dumpStep = true;
143 time = t;
144 getOutputVars(theVars);
145 MB(t, y);
146 getResults(theVars);
147
148 if (theVars != inputVars)
149 inputVars->deepcopy(theVars);
150 IniModelCom(inputVars); // Reset connections
151}

162arr Driver::run(const bool continuingRun) {
163
164 uint count = 0;
165 _firstPass = true;
166
167 while (count < 10){
168 maxStep = 20. * step;
169
170 setup(nullptr, continuingRun);
171
172 sunindextype N = static_cast<long>(constraints.size());
173 N_Vector y;
174#ifdef SUNDIALS_CONTEXT_REQUIRED
175 y = N_VNew_Serial(N, context());
176#else // SUNDIALS_CONTEXT_REQUIRED
177 y = N_VNew_Serial(N);
178#endif // SUNDIALS_CONTEXT_REQUIRED
179 sunrealtype* y_ptr = N_VGetArrayPointer(y);
180
181 for (std::size_t i = 0; i < constraints.size(); i++)
182 y_ptr[i] = constraints[i];
183 realtype t0 = start;
184
185 CVodeMem *cmem = nullptr;
186 try {
187 cmem = &CVodeMem::create();
188 cmem->cvode_mem_init(this, t0, y);
189 } catch(...) {
190 std::exception_ptr eptr = std::current_exception();
191 N_VDestroy(y);
192 std::rethrow_exception(eptr);
193 }
194
195 data->drv = this;
196
197#ifdef SUNDIALS_CONTEXT_REQUIRED
198 SUNMatrix A = SUNDenseMatrix(N, N, context());
199 SUNNonlinearSolver NLS = SUNNonlinSol_Newton(y, context());
200 SUNLinearSolver LS = SUNLinSol_Dense(y, A, context());
201#else // SUNDIALS_CONTEXT_REQUIRED
202 SUNMatrix A = SUNDenseMatrix(N, N);
203 SUNNonlinearSolver NLS = SUNNonlinSol_Newton(y);
204 SUNLinearSolver LS = SUNLinSol_Dense(y, A);
205#endif // SUNDIALS_CONTEXT_REQUIRED
206
207 try {
208 if (CVodeSetNonlinearSolver(cvode_mem, NLS) != CV_SUCCESS) {
209 std::cout << "CVodeSetNonlinearSolver failed" << std::endl;
210 throw std::runtime_error("CVodeSetNonlinearSolver failed");
211 }
212 if (CVodeSetLinearSolver(cvode_mem, LS, A) != CV_SUCCESS) {
213 std::cout << "CVodeSetLinearSolver failed" << std::endl;
214 throw std::runtime_error("CVodeSetLinearSolver failed");
215 }
216 } catch(...) {
217 std::exception_ptr eptr = std::current_exception();
218 SUNNonlinSolFree(NLS);
219 SUNLinSolFree(LS);
220 SUNMatDestroy(A);
221 N_VDestroy(y);
222 std::rethrow_exception(eptr);
223 }
224
225 realtype t = start;
226 bool runOK = true;
227 realtype tout = start + step;
228 while (t < endtime) {
229 if (CVode(cvode_mem, tout, y, &t, CV_NORMAL) != CV_SUCCESS) {
230 std::cout << "CVode failed at t=" << tout << " " << t << std::endl;
231 runOK = false;
232 break;
233 }
234 tout += step;
235 if (tout > endtime)
236 tout = endtime;
237 }
238 if (runOK) {
239 teardown(t, y);
240 }
241
242 SUNNonlinSolFree(NLS);
243 SUNLinSolFree(LS);
244 SUNMatDestroy(A);
245 N_VDestroy(y);
246 intermediateRes = nullptr;
247 if (currentVars) {
248 delete currentVars;
249 currentVars = nullptr;
250 }
251 if (runOK)
252 return results;
253
254 count++;
255 step = initialStep / (count + 1);
256 _firstPass = false;
257 std::cout << "Retrying with smaller step size: " << step << std::endl;
258 }
259 throw std::runtime_error("No valid solution found");
260}

414void Driver::_dump(realtype t, ValueSet_t* con) {
415 if (!_dumpStep) return;
416 if (t == 0 && !fname_vars_init.empty()) {
417 dump(fname_vars_init, nullptr, con, true);
418 } else if (_lastStep && !fname_vars_last.empty()) {
419 dump(fname_vars_last, nullptr, con, false);
420 }
421 if (!fname_vars_step.empty()) {
422 if (t >= endtime) return;
423 std::ostringstream tss;
424 tss.precision(6);
425 tss << std::fixed << t;
426 std::string tstr = tss.str();
427 tstr = std::string(13 - tstr.length(), '0') + tstr;
428 std::string ifile = fname_vars_step + tstr + ".txt";
429 dump(ifile, nullptr, con, false);
430 }
431 if (fname_vars_step.empty())
432 _dumpStep = false;
433}
