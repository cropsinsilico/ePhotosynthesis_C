65void DynaPSDriver::setup_connections(Variables* theVars) {
66
67 // The combination of BF and FI model
68 theVars->BF_FI_com = true;
69
70 // This is a variable indicating whether the PR model is actually need to be combined with PS or not. If 1 then means combined; 0 means not.
71 theVars->PR_PS_com = true;
72
73 // 1 means that the overall EPS model is used. 0 means partial model of FIBF is used.
74 theVars->FIBF_PSPR_com = true;
75
76 // A global variable to indicate whether the RuACT is run by itself or combined with others.
77 // langmm: This is true in the MATLAB code, but was false in the
78 // original translation to C++
79 theVars->RuACT_EPS_com = true; // Since this is run within this program, it is combinbed, therefore, it is assigned value true, otherwise, assign value false.
80
81 // This is the connection between Redox and RA.
82 theVars->RedoxReg_RA_com = false; // This means that the connection is not provided there.
83
84 theVars->XanCycle_BF_com = true;
85
86 theVars->EPS_SUCS_com = true;
87
88 // This is a variable indicating whether the PSPR model is actually need to be combined with SUCS or not. If 1 then means combined; 0 means not.
89 theVars->PSPR_SUCS_com = true;
90
91}

137void DynaPSDriver::getResults(Variables* theVars) {
138
139 if (!theVars) theVars = currentVariables();
140
141 double PSIIabs = theVars->FI_Vel.vP680_d;
142 double PSIabs = theVars->BF_Vel.Vbf11;
143 double CarbonRate = theVars->RuACT_Vel.v6_1 * theVars->AVR;
144 double VPR = theVars->RuACT_Vel.v6_2 * theVars->AVR;
145 double Vpgasink = theVars->SUCS_Vel.vpga_use * theVars->AVR;
146 double VStarch = (theVars->PS_Vel.v23 - theVars->PS_Vel.v25) * theVars->AVR;
147 double Vsucrose = theVars->SUCS_Vel.vdhap_in * theVars->AVR;
148 const double CO2AR = TargetFunVal(theVars);
149
150 results = zeros(8);
151
152 results[0] = PSIIabs;
153 results[1] = PSIabs;
154 //Resulta(3)=PSIabs2(row);
155 results[2] = CarbonRate;
156 results[3] = VPR;
157 results[4] = Vpgasink;
158 results[5] = Vsucrose;
159 results[6] = VStarch;
160 results[7] = CO2AR;
161
162 if(theVars->record) {
163 makeFluxTR(theVars);
164 }
165
166}
